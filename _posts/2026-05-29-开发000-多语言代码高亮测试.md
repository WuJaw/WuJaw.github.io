## STM32 USART 中断接收（C）

```c
#include "stm32f4xx_hal.h"

#define RX_BUF_SIZE 64

static uint8_t rx_buf[RX_BUF_SIZE];
static uint8_t rx_idx = 0;

/* 包头常量 */
#define HEADER_0  0x87
#define HEADER_1  0xF0

void HAL_UART_RxCpltCallback(UART_HandleTypeDef *huart)
{
    uint8_t ch;

    if (huart->Instance != USART2) {
        return;
    }

    ch = rx_buf[rx_idx];

    /* 逐字节判断 —— 不用状态机 */
    if (rx_idx == 0 && ch != HEADER_0) {
        rx_idx = 0;
        /* 不是合法包头，重新等 */
        HAL_UART_Receive_IT(&huart2, rx_buf + rx_idx, 1);
        return;
    }

    rx_idx++;

    if (rx_idx == 2) {
        if (rx_buf[0] == HEADER_0 && rx_buf[1] == HEADER_1) {
            /* 包头校验通过，继续收后续数据 */
        } else {
            rx_idx = 0;
        }
    }

    HAL_UART_Receive_IT(&huart2, rx_buf + rx_idx, 1);
}

void uart_init(void)
{
    __HAL_RCC_GPIOA_CLK_ENABLE();
    __HAL_RCC_USART2_CLK_ENABLE();

    GPIO_InitTypeDef gpio = {0};
    gpio.Pin       = GPIO_PIN_2 | GPIO_PIN_3;
    gpio.Mode      = GPIO_MODE_AF_PP;
    gpio.Pull      = GPIO_NOPULL;
    gpio.Speed     = GPIO_SPEED_FREQ_HIGH;
    gpio.Alternate = GPIO_AF7_USART2;
    HAL_GPIO_Init(GPIOA, &gpio);

    UART_HandleTypeDef huart2 = {0};
    huart2.Instance          = USART2;
    huart2.Init.BaudRate     = 115200;
    huart2.Init.WordLength   = UART_WORDLENGTH_8B;
    huart2.Init.StopBits     = UART_STOPBITS_1;
    huart2.Init.Parity       = UART_PARITY_NONE;
    huart2.Init.Mode         = UART_MODE_TX_RX;
    huart2.Init.HwFlowCtl    = UART_HWCONTROL_NONE;
    HAL_UART_Init(&huart2);

    /* 启动中断接收 */
    HAL_UART_Receive_IT(&huart2, rx_buf, 1);
}
```

## BW16 AT 指令驱动框架（C）

```c
#include <stdint.h>
#include <string.h>
#include <stdio.h>

#define BW16_CMD_TIMEOUT_MS  5000
#define BW16_RX_BUF_SIZE     256

typedef enum {
    BW16_OK,
    BW16_ERROR,
    BW16_TIMEOUT,
    BW16_BUSY,
} bw16_status_t;

static char bw16_rx[BW16_RX_BUF_SIZE];
static uint16_t bw16_rx_len;
static volatile uint8_t bw16_ready;

int bw16_send_at(const char *cmd, const char *expect, uint32_t timeout_ms)
{
    bw16_ready = 0;
    bw16_rx_len = 0;
    memset(bw16_rx, 0, BW16_RX_BUF_SIZE);

    /* 发送 AT 指令 */
    uart_send_string(cmd);
    uart_send_string("\r\n");

    /* 等待响应 */
    uint32_t tick = HAL_GetTick();
    while (!bw16_ready) {
        if (HAL_GetTick() - tick > timeout_ms) {
            return BW16_TIMEOUT;
        }
    }

    if (strstr(bw16_rx, "ERROR")) {
        return BW16_ERROR;
    }
    if (expect && !strstr(bw16_rx, expect)) {
        return BW16_ERROR;
    }
    return BW16_OK;
}

bw16_status_t bw16_wifi_connect(const char *ssid, const char *pwd)
{
    char buf[128];
    snprintf(buf, sizeof(buf), "AT+CWJAP=\"%s\",\"%s\"", ssid, pwd);
    return bw16_send_at(buf, "OK", BW16_CMD_TIMEOUT_MS);
}
```

## CH592 BLE 广播配置（C）

```c
#include "CH59x_common.h"

#define DEVICE_NAME "CH592-BLE"

static uint8_t adv_data[] = {
    0x02, 0x01, 0x06,                         /* Flags: LE General Discoverable */
    0x09, 0x08, 'C','H','5','9','2','-','B','L','E',  /* 设备名 */
    0x03, 0x02, 0xF0, 0xFF,                   /* 16-bit UUID */
};

void ble_adv_init(void)
{
    GAPRole_SetParameter(GAPROLE_ADVERT_ENABLED, sizeof(uint8_t), &(uint8_t){TRUE});

    /* 设置广播数据 */
    GAP_UpdateAdvertisingData(GAPROLE_ADVERT_STATIC,
                              TRUE,
                              sizeof(adv_data),
                              adv_data);

    /* 100ms 广播间隔 */
    uint16_t interval = 160; /* 160 * 0.625ms = 100ms */
    GAP_SetParamValue(TGAP_DISC_ADV_INT_MIN, interval);
    GAP_SetParamValue(TGAP_DISC_ADV_INT_MAX, interval);
}
```

## 图片路径转换脚本（Python）

```python
#!/usr/bin/env python3
"""将 Markdown 中图片路径从 Typora 格式转为 Jekyll 网页格式."""

import os
import re
import glob
import shutil

ASSETS_DIR = "assets"
POSTS_DIR = "_posts"

IMG_TAG_RE = re.compile(
    r'!\[(.*?)\]\((.*?)\)'
)

def find_image(filename):
    """在 assets 目录及全局搜索图片文件."""
    assets_path = os.path.join(ASSETS_DIR, filename)
    if os.path.exists(assets_path):
        return assets_path
    # 全局搜索
    for root, dirs, files in os.walk("."):
        if filename in files:
            return os.path.join(root, filename)
    return None

def fix_post(filepath):
    with open(filepath, 'r', encoding='utf-8') as f:
        content = f.read()

    def replace_img(m):
        alt = m.group(1)
        src = m.group(2)

        # 已经是正确格式
        if src.startswith('/assets/'):
            return m.group(0)
        if src.startswith('<img '):
            return m.group(0)

        # 提取文件名
        basename = os.path.basename(src)
        target = find_image(basename)

        if target and not target.startswith(ASSETS_DIR + os.sep):
            shutil.copy2(target, os.path.join(ASSETS_DIR, basename))

        return f'<img src="/assets/{basename}" alt="{alt}">'

    new_content = IMG_TAG_RE.sub(replace_img, content)

    with open(filepath, 'w', encoding='utf-8') as f:
        f.write(new_content)

def main():
    posts = glob.glob(os.path.join(POSTS_DIR, "*.md"))
    for p in posts:
        fix_post(p)
        print(f"OK: {p}")

if __name__ == "__main__":
    main()
```

## Makefile 编译示例

```makefile
# CH592 BLE 项目 Makefile

TARGET    := ch592_ble
MCU       := CH592F
TOOLCHAIN := riscv-none-embed

CC       := $(TOOLCHAIN)-gcc
OBJCOPY  := $(TOOLCHAIN)-objcopy
SIZE     := $(TOOLCHAIN)-size

CFLAGS   := -march=rv32imac -mabi=ilp32 \
            -Os -ffunction-sections -fdata-sections \
            -Wall -Wextra -Werror

LDFLAGS  := -nostartfiles -Wl,--gc-sections \
            -T ch592f.ld

SRCS     := $(wildcard src/*.c)
OBJS     := $(SRCS:.c=.o)

all: $(TARGET).bin

$(TARGET).elf: $(OBJS)
	$(CC) $(CFLAGS) $(LDFLAGS) -o $@ $^

%.o: %.c
	$(CC) $(CFLAGS) -c -o $@ $<

$(TARGET).bin: $(TARGET).elf
	$(OBJCOPY) -O binary $< $@
	$(SIZE) $<

flash: $(TARGET).bin
	wchisp flash $<

clean:
	rm -f src/*.o $(TARGET).elf $(TARGET).bin

.PHONY: all flash clean
```

## BLE 数据解析（Python）

```python
"""解析 BLE Manufacturer Data 中的温湿度传感器数据."""

import struct

def parse_manufacturer_data(data: bytes) -> dict:
    """
    数据格式（大端）：
    [0:2]   Company ID
    [2]     Sensor Type (0x01 = 温湿度)
    [3:5]   温度 (int16, 单位 0.01°C)
    [5:7]   湿度 (uint16, 单位 0.01%)
    """
    if len(data) < 7:
        raise ValueError(f"数据太短: {len(data)} bytes")

    company_id = struct.unpack(">H", data[0:2])[0]
    sensor_type = data[2]
    temp_raw = struct.unpack(">h", data[3:5])[0]
    humi_raw = struct.unpack(">H", data[5:7])[0]

    return {
        "company_id":  f"0x{company_id:04X}",
        "sensor_type": sensor_type,
        "temperature": round(temp_raw * 0.01, 2),
        "humidity":    round(humi_raw * 0.01, 2),
    }

# 测试数据
test_data = bytes([0x12, 0x34, 0x01, 0x0C, 0xE4, 0x16, 0x4E])

result = parse_manufacturer_data(test_data)
for k, v in result.items():
    print(f"  {k}: {v}")
```
